# Import Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `import <format> <file> [--write]`, which merges a bank's hand-exported file into `<account>/import/<year>.json`.

**Architecture:** A pure reader in `sync/src/import/` turns bytes into checked text, text into a table, and a table into one batch per account. The command in `sync/src/commands/import.ts` feeds each batch through the existing `merge` and the existing store. `shared` gains two configuration rules and trimming in the view.

**Tech Stack:** Node 24 running TypeScript directly, `node:test`, `TextDecoder`, no dependencies.

**Spec:** [docs/specs/2026-10-13-import.md](../specs/2026-10-13-import.md). Read it first; this plan argues from it.

## Global Constraints

- **No real data, and no bank is named.** Every header, IBAN, name and amount in code and tests is invented. Invented IBANs use check digits `00`. Tests use the made-up format that `aFormat()` and `anImportRaw()` in `shared/testing/factories.ts` describe.
- **Messages never repeat a cell, a header name, a path or a command-line argument.** They name rows by number, fields by the format's field name, and accounts and formats by configured key.
- **No dependencies are added**, runtime or development.
- **Erasable TypeScript only**: no `enum`, no parameter properties. Relative imports end in `.ts`. Type-only imports use `import type`.
- **No silent error handling**: no empty `catch`, no default that hides missing data.
- **Run tests with `npm test`** (it pins `TZ=Europe/Amsterdam`). To run one file: `TZ=Europe/Amsterdam node --test <path>`.
- **The pre-commit hook runs typecheck, lint, format check and the whole suite.** Never pass `--no-verify`. Run `npm run format` before each commit.
- **Commits are conventional**: `type(scope): message`, scope `shared` or `sync`, left out when a change spans both.
- **`no-control-regex` is on.** Never write a control character into a regular expression; test code points with numbers instead.
- Work in the worktree at `.claude/worktrees/import`. `npm ci` has been run there.

```ts
  if (!isObject(format)) return ["is not an object"];
  const problems: string[] = unknownFields(format, FORMAT_FIELDS);

  };

    const value = format[name];
    if (value === undefined && !required) return;
    if (!isColumn(value)) problems.push(`${name} must name a column`);
    else within(name, value, ["column"]);
```

```ts
import { SyncError } from "../errors.ts";

// A file that is refused. Its message is composed here from row numbers, field names and
// keys, and never repeats a header or a cell, so it may be printed.
export class ImportError extends SyncError {
  override name = "ImportError";
}

const PROBLEM_LIMIT = 20;

// One refusal that lists what was found: enough to see what is wrong with a file, and
// not a line for every row of a file read with the wrong format.
export const reject = (problems: readonly string[]): ImportError => {
  const shown = problems.slice(0, PROBLEM_LIMIT);
  const more = problems.length - shown.length;
  return new ImportError([...shown, ...(more > 0 ? [`and ${more} more`] : [])].join("\n"));
};
```

Create `sync/src/import/decode.ts`:

```ts
// Bytes to text, by the encoding the import format declares. The decoded text is stored
// for good, and a wrong single-byte encoding still decodes, to the wrong letters. So every
// mismatch that can be detected refuses the file.
import type { ImportFormat } from "@money/shared";

import { ImportError } from "./error.ts";

type Encoding = ImportFormat["encoding"];

const UTF8_MARK = [0xef, 0xbb, 0xbf];

const startsWith = (bytes: Uint8Array, mark: readonly number[]): boolean =>
  mark.every((byte, index) => bytes[index] === byte);

// The text, or null when the bytes are not UTF-8.
const strictUtf8 = (bytes: Uint8Array): string | null => {
  try {
    return new TextDecoder("utf-8", { fatal: true, ignoreBOM: true }).decode(bytes);
  } catch (error) {
    if (!(error instanceof TypeError)) throw error;
    return null;
  }
};

// Anything below a space other than tab and the line endings, DEL, and U+0080 to U+009F.
// The last range is what a byte with no meaning in the declared encoding decodes to.
const isControl = (code: number): boolean =>
  (code < 0x20 && code !== 0x09 && code !== 0x0a && code !== 0x0d) ||
  (code >= 0x7f && code <= 0x9f);

const hasControl = (text: string): boolean => {
  for (let index = 0; index < text.length; index += 1) {
    if (isControl(text.charCodeAt(index))) return true;
  }
  return false;
};

const beyondAscii = (text: string): boolean => /[\u0080-￿]/.test(text);

export const decode = (bytes: Uint8Array, encoding: Encoding): string => {
  if (startsWith(bytes, [0xff, 0xfe]) || startsWith(bytes, [0xfe, 0xff])) {
    throw new ImportError("the file starts with a UTF-16 byte order mark, which no format reads");
  }
  const marked = startsWith(bytes, UTF8_MARK);
  let text: string;
  if (encoding === "utf-8") {
    // The mark is not part of the first header name.
    const decoded = strictUtf8(marked ? bytes.subarray(UTF8_MARK.length) : bytes);
    if (decoded === null) {
      throw new ImportError("the file is not valid UTF-8, but its format declares utf-8");
    }
    text = decoded;
  } else {
    if (marked) {
      throw new ImportError(
        `the file starts with a UTF-8 byte order mark, but its format declares ${encoding}`,
      );
    }
    // Real single-byte text almost never forms valid multi-byte UTF-8 by accident.
    const asUtf8 = strictUtf8(bytes);
    if (asUtf8 !== null && beyondAscii(asUtf8)) {
      throw new ImportError(
        `the file is valid UTF-8 with characters beyond ASCII, but its format declares ${encoding}`,
      );
    }
    text = new TextDecoder(encoding).decode(bytes);
  }
  if (hasControl(text)) {
    throw new ImportError(
      `the file holds a control character once read as ${encoding}, so the encoding is probably another`,
    );
  }
  return text;
};
```

```ts
// Text to a header and rows of cells. Quoting follows RFC 4180: a cell that starts with a
// double quote runs to the closing one, and may hold the delimiter, a line break and a
// doubled quote. Anything else that is not a plain table is refused, never repaired: what
// is read here is stored for good.
import { ImportError, reject } from "./error.ts";

export type Table = { header: string[]; rows: string[][] };

const cells = (count: number): string => `${count} ${count === 1 ? "cell" : "cells"}`;

const split = (text: string, delimiter: string): string[][] => {
  const rows: string[][] = [];
  let row: string[] = [];
  let cell = "";
  // start: nothing of the cell read yet. plain: inside an unquoted cell. open: inside
  // quotes. closed: just past the closing quote.
  let state: "start" | "plain" | "open" | "closed" = "start";
  const here = (): string => `row ${rows.length + 1}`;
  const endCell = (): void => {
    row.push(cell);
    cell = "";
    state = "start";
  };
  const endRow = (): void => {
    endCell();
    rows.push(row);
    row = [];
  };

  for (let index = 0; index < text.length; index += 1) {
    const character = text[index];
    if (state === "open") {
      if (character !== '"') cell += character;
      else if (text[index + 1] === '"') {
        cell += '"';
        index += 1;
      } else state = "closed";
    } else if (character === delimiter) endCell();
    else if (character === "\n") endRow();
    else if (character === "\r" && text[index + 1] === "\n") {
      index += 1;
      endRow();
    } else if (state === "closed") {
      throw new ImportError(`${here()} has text after a closing quote`);
    } else if (character === '"') {
      if (state !== "start") {
        throw new ImportError(`${here()} has a quote inside a cell that is not quoted`);
      }
      state = "open";
    } else {
      cell += character;
      state = "plain";
    }
  }
  if (state === "open") throw new ImportError(`${here()} has a quote that is never closed`);
  // A line ending at the end of the file ends the last row and starts no new one.
  if (state !== "start" || row.length > 0) endRow();
  return rows;
};

export const readDelimited = (text: string, delimiter: string): Table => {
  const all = split(text, delimiter);
  if (all.length === 0) throw new ImportError("the file is empty");
  const [header, ...rows] = all;

  const problems: string[] = [];
  const firstAt = new Map<string, number>();
  header.forEach((name, index) => {
    if (name === "") problems.push(`column ${index + 1} of the header has no name`);
    const earlier = firstAt.get(name);
    if (earlier === undefined) firstAt.set(name, index);
    else if (name !== "") {
      problems.push(`columns ${earlier + 1} and ${index + 1} of the header have the same name`);
    }
  });
  if (rows.length === 0) problems.push("the file has a header and no rows");
  rows.forEach((row, index) => {
    if (row.length !== header.length) {
      problems.push(
        `row ${index + 2} has ${cells(row.length)}, and the header has ${header.length}`,
      );
    }
  });
  if (problems.length > 0) throw reject(problems);
  return { header, rows };
};
```

```ts
// A table to one batch per account, ready for merge. A row's identifier and booking date
// are read here, once, and its amount and currency are checked. Nothing else of a row is
// judged: `raw` is stored as it is, and the view interprets it.
import { CURRENCY_CODE, namedColumns, parseAmount, parseDate } from "@money/shared";
import type { Config, ImportFormat, ImportRaw, Incoming } from "@money/shared";

import { plainIban } from "../iban.ts";
import type { Table } from "./delimited.ts";
import { reject } from "./error.ts";

export type Batch = { account: string; incoming: Incoming[] };

// What an export file calls an account, to its key: the import_id exactly, or the IBAN
// whatever its spacing and case. The configuration refuses two accounts that share either.
const accountFinder = (accounts: Config["accounts"]): ((cell: string) => string | undefined) => {
  const byImportId = new Map<string, string>();
  const byIban = new Map<string, string>();
  for (const [key, account] of Object.entries(accounts)) {
    if (account.import_id !== undefined) byImportId.set(account.import_id, key);
    if (account.iban !== undefined) byIban.set(plainIban(account.iban), key);
  }
  return (cell) => byImportId.get(cell) ?? byIban.get(plainIban(cell));
};

export const readRows = (
  table: Table,
  format: ImportFormat,
  accounts: Config["accounts"],
): Batch[] => {
  const { header, rows } = table;
  const missing = namedColumns(format)
    .filter(([, column]) => !header.includes(column))
    .map(([field]) => `the header lacks the column that the format names as ${field}`);
  if (missing.length > 0) throw reject(missing);

  const at = {
    account: header.indexOf(format.account.column),
    id: header.indexOf(format.id.column),
    date: header.indexOf(format.date.column),
    amount: header.indexOf(format.amount.column),
    currency: header.indexOf(format.currency.column),
  };
  const find = accountFinder(accounts);
  const problems: string[] = [];
  const batches = new Map<string, Incoming[]>();
  // Per account, the row each identifier was first seen on.
  const firstRow = new Map<string, Map<string, number>>();

  rows.forEach((cells, index) => {
    const number = index + 2;
    const before = problems.length;
    const account = find(cells[at.account]);
    if (account === undefined)
      problems.push(`row ${number} is for an account that is not configured`);
    const id = cells[at.id];
    if (id === "") problems.push(`row ${number} has no identifier`);
    const date = parseDate(cells[at.date], format.date.format);
    if (date === null) problems.push(`row ${number} has a date that cannot be read`);
    const currency = cells[at.currency];
    if (!CURRENCY_CODE.test(currency)) {
      problems.push(`row ${number} has a currency that cannot be read`);
    } else if (parseAmount(cells[at.amount], format.amount.decimal, currency) === null) {
      problems.push(`row ${number} has an amount that cannot be read`);
    }
    if (account !== undefined && id !== "") {
      const seen = firstRow.get(account) ?? new Map<string, number>();
      firstRow.set(account, seen);
      const earlier = seen.get(id);
      // Stricter than merge, which accepts an identical repeat: a file that lists one
      // transaction twice is not the export it claims to be.
      if (earlier !== undefined)
        problems.push(`row ${number} repeats the identifier of row ${earlier}`);
      else seen.set(id, number);
    }
    if (problems.length > before || account === undefined || date === null) return;

    const raw: ImportRaw = Object.fromEntries(header.map((name, column) => [name, cells[column]]));
    const batch = batches.get(account) ?? [];
    batches.set(account, batch);
    batch.push({ id, date, raw });
  });

  if (problems.length > 0) throw reject(problems);
  return [...batches]
    .toSorted(([a], [b]) => (a < b ? -1 : 1))
    .map(([account, incoming]) => ({ account, incoming }));
};
```

```ts
export const renderExport = (
  header: readonly string[],
  rows: readonly (readonly string[])[],
  options: {
    delimiter?: string;
    eol?: string;
    finalEol?: boolean;
    quote?: (cell: string) => boolean;
  } = {},
): string => {
  const { delimiter = ";", eol = "\r\n", finalEol = true, quote = () => false } = options;
  const cell = (text: string): string =>
    text.includes(delimiter) || /["\r\n]/.test(text) || quote(text)
      ? `"${text.replaceAll('"', '""')}"`
      : text;
  const lines = [header, ...rows].map((cells) => cells.map(cell).join(delimiter));
  return lines.join(eol) + (finalEol ? eol : "");
};
```

```ts
// The characters of a file beyond ASCII, counted. windows-1252 and iso-8859-15 differ in
// eight characters and no check can tell them apart, so the person importing is shown what
// the declared encoding made of the file: a stray "¤" where a "€" belongs is visible here.
export type CharacterCount = { character: string; count: number };

export const nonAscii = (text: string): CharacterCount[] => {
  const counts = new Map<string, number>();
  // Iterating a string yields whole code points, so a pair of surrogates is one character.
  for (const character of text) {
    if ((character.codePointAt(0) ?? 0) > 0x7f) {
      counts.set(character, (counts.get(character) ?? 0) + 1);
    }
  }
  return [...counts]
    .map(([character, count]) => ({ character, count }))
    .toSorted((a, b) => (a.character.codePointAt(0) ?? 0) - (b.character.codePointAt(0) ?? 0));
};

export const describeCharacter = ({ character, count }: CharacterCount): string => {
  const code = (character.codePointAt(0) ?? 0).toString(16).toUpperCase().padStart(4, "0");
  return `  U+${code} ${character} ${count}`;
};
```

```ts
// Merges a file exported from a bank's website into the archive. It checks by default and
// writes only when asked: an import is permanent, like everything in the archive.
import { readFile } from "node:fs/promises";

import { merge, renderYearFiles } from "@money/shared";
import type { ArchiveRecord, Target } from "@money/shared";
import { readAccountSource, recoverAccountSource, writeYearFiles } from "@money/shared/node";

import type { Context } from "../context.ts";
import { SyncError } from "../errors.ts";
import { describeCharacter, nonAscii } from "../import/characters.ts";
import { decode } from "../import/decode.ts";
import { readDelimited } from "../import/delimited.ts";
import { reject } from "../import/error.ts";
import { readRows } from "../import/rows.ts";
import { timestampOf } from "../time.ts";

const UNREADABLE = ["ENOENT", "EISDIR", "ENOTDIR", "EACCES"];

// The path is typed by a person and may name anything, so the message holds only the code.
const readBytes = async (path: string): Promise<Uint8Array> => {
  try {
    return await readFile(path);
  } catch (error) {
    const code = error instanceof Error ? (error as NodeJS.ErrnoException).code : undefined;
    if (code === undefined || !UNREADABLE.includes(code)) throw error;
    throw new SyncError(`the file to import cannot be read (${code})`);
  }
};

type Merged = {
  account: string;
  rows: number;
  added: number;
  revised: number;
  unchanged: number;
  from: string;
  to: string;
  records: ArchiveRecord[];
};

export const runImport = async (
  context: Context,
  formatKey: string,
  path: string,
  write: boolean,
): Promise<number> => {
  const { settings, config, deps, log } = context;
  if (!Object.hasOwn(config.imports, formatKey)) {
    const keys = Object.keys(config.imports).toSorted();
    throw new SyncError(
      `no import format is configured under that key. Configured: ${keys.length > 0 ? keys.join(", ") : "none"}`,
    );
  }
  const format = config.imports[formatKey];
  const text = decode(await readBytes(path), format.encoding);
  const batches = readRows(readDelimited(text, format.delimiter), format, config.accounts);

  // Every account is merged before any is written: a file is accepted or refused whole.
  const now = timestampOf(deps.clock());
  const merged: Merged[] = [];
  const problems: string[] = [];
  for (const { account, incoming } of batches) {
    // A check writes nothing, so it leaves a doubled record to stop it below.
    if (write) await recoverAccountSource(settings.dataDir, account, "import");
    const stored = await readAccountSource(settings.dataDir, account, "import");
    // merge would keep the old format key over a raw that it does not describe.
    const ids = new Set(incoming.map((item) => item.id));
    const elsewhere = stored.filter(
      (record) => record.source === "import" && record.format !== formatKey && ids.has(record.id),
    ).length;
    if (elsewhere > 0) {
      problems.push(
        `${account}: ${elsewhere} of the file's identifiers are stored under another import format`,
      );
      continue;
    }
    const target: Target = { account, source: "import", format: formatKey };
    const result = merge(stored, incoming, target, now);
    const dates = incoming.map((item) => item.date).toSorted();
    merged.push({
      account,
      rows: incoming.length,
      added: result.added.length,
      revised: result.revised.length,
      unchanged: result.unchanged,
      from: dates[0],
      to: dates[dates.length - 1],
      records: result.records,
    });
  }
  if (problems.length > 0) throw reject(problems);

  for (const item of merged) {
    log("import-account", {
      account: item.account,
      rows: item.rows,
      new: item.added,
      changed: item.revised,
      unchanged: item.unchanged,
      from: item.from,
      to: item.to,
    });
  }
  // The one thing printed that is not a key, a date or a count: single characters, so that
  // a wrong encoding can be seen before anything is stored.
  const characters = nonAscii(text);
  log("import-characters", { distinct: characters.length });
  for (const found of characters) deps.out(describeCharacter(found));

  if (write) {
    for (const item of merged) {
      const { written } = await writeYearFiles(
        settings.dataDir,
        item.account,
        "import",
        renderYearFiles(item.records),
      );
      log("import-written", { account: item.account, files: written.length });
    }
  }
  log("import-checked", {
    format: formatKey,
    accounts: merged.length,
    rows: merged.reduce((sum, item) => sum + item.rows, 0),
    written: write ? "yes" : "no",
  });
  if (!write) deps.out("Nothing was written. Run the same command with --write to import.");
  return 0;
};
```

```ts
  return Array.from({ length: ROWS }, (_, index) => {

    // Twenty days around a year boundary.
    const date = addDays("2024-12-22", random.int(0, 20));
    const cents = random.int(1, 99_999) * (random.chance(0.6) ? -1 : 1);
    const sign = cents < 0 ? "-" : random.chance(0.5) ? "+" : "";
    const amount = `${sign}${Math.floor(Math.abs(cents) / 100)},${String(Math.abs(cents) % 100).padStart(2, "0")}`;

      "Text 2": "",
      "Text 3": "",

        revised: false,
      },
    };
  });

    const world = await makeWorld({
      config: { ...worldConfig(), imports: { [FORMAT]: aFormat() } },
    });
    t.after(world.cleanup);
    const random = makeRandom(seed * 7919);
    const rows = generate(seed).filter(
      // One account may not hold an identifier twice; the counter gives each id two rows.
      (row, index, all) =>
        all.findIndex(
          (other) =>
            other.expected.account === row.expected.account &&
            other.expected.id === row.expected.id,
        ) === index,
    );

    // Three files that together hold every row, three more that overlap them at random,
    // and one of the six again.
    const a = random.int(1, rows.length - 2);
    const b = random.int(a + 1, rows.length - 1);
    const ranges: [number, number][] = [
      [0, a],
      [a, b],
      [b, rows.length],
    ];
    for (let extra = 0; extra < 3; extra += 1) {
      const from = random.int(0, rows.length - 1);
      ranges.push([from, random.int(from + 1, rows.length)]);
    }
    const schedule = random.shuffle(ranges);
    schedule.push(random.pick(ranges));
    seen.repeated += 1;

    for (const [index, [from, to]] of schedule.entries()) {
      const slice = random.shuffle(rows.slice(from, to));
      const finalEol = random.chance(0.7);
      const text = renderExport(
        HEADER,
        slice.map((row) => row.cells),
        {
          eol: random.pick(["\r\n", "\n"]),
          finalEol,
          quote: () => random.chance(0.4),
        },
      );
      seen.files += 1;
      if (!finalEol) seen.withoutFinalEol += 1;
      if (slice.some((row) => row.cells.some((cell) => cell.includes(";"))))
        seen.quotedDelimiter += 1;
      if (slice.some((row) => row.cells.some((cell) => cell.includes("\n")))) seen.lineBreak += 1;

      const path = join(dirname(world.dataDir), `export-${index}.csv`);
      await writeFile(path, text);
      world.at(`2026-03-${String(index + 1).padStart(2, "0")}T10:00:00Z`);
      world.lines.length = 0;
      const code = await world.run("import", FORMAT, path, "--write");
      assert.equal(code, 0, `seed ${seed}, file ${index}: ${world.lines.join(" | ")}`);
    }

    const actual = (await readArchive(world.dataDir)).map((record) => ({
      ...normalise(record, world.config.imports),
      first_seen: "",
    }));
    assert.deepEqual(inOrder(actual), inOrder(rows.map((row) => row.expected)), `seed ${seed}`);
  });
}

test("the schedules reach what the property exists for", () => {
```
